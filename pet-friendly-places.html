const GOOGLE_PLACES_URL = "https://places.googleapis.com/v1/places:searchNearby"; const DEFAULT_RADIUS = 5000; const MAX_RADIUS = 25000; const MAX_RESULTS = 20; const CATEGORY_TYPES = { All: [ "cafe", "restaurant", "hotel", "veterinary_care", "pet_store" ], Cafe: ["cafe"], Restaurant: ["restaurant"], Hotel: ["hotel"], Veterinary: ["veterinary_care"], "Pet Shop": ["pet_store"] }; function setCors(res) { res.setHeader("Access-Control-Allow-Origin", "*"); res.setHeader("Access-Control-Allow-Methods", "POST, OPTIONS"); res.setHeader("Access-Control-Allow-Headers", "Content-Type"); } function json(res, status, payload) { setCors(res); res.status(status).json(payload); } function toNumber(value) { const number = Number(value); return Number.isFinite(number) ? number : null; } function clampRadius(value) { const parsed = toNumber(value); if (!parsed) { return DEFAULT_RADIUS; } return Math.min( MAX_RADIUS, Math.max(100, Math.round(parsed)) ); } function normalizeCategory(value) { const category = String(value || "All").trim(); if (Object.prototype.hasOwnProperty.call(CATEGORY_TYPES, category)) { return category; } return "All"; } function buildFieldMask() { return [ "places.id", "places.displayName", "places.formattedAddress", "places.location", "places.primaryType", "places.primaryTypeDisplayName", "places.types", "places.rating", "places.userRatingCount", "places.googleMapsUri", "places.websiteUri", "places.nationalPhoneNumber", "places.internationalPhoneNumber", "places.regularOpeningHours", "places.currentOpeningHours", "places.photos", "places.businessStatus", "places.allowsDogs" ].join(","); } function normalizeGoogleCategory(place) { const primary = String(place.primaryType || "").toLowerCase(); const types = Array.isArray(place.types) ? place.types.map(item => String(item).toLowerCase()) : []; const all = [primary, ...types]; if ( all.includes("cafe") || all.includes("coffee_shop") ) { return "Cafe"; } if ( all.includes("restaurant") || all.includes("bar") || all.includes("pub") ) { return "Restaurant"; } if ( all.includes("hotel") || all.includes("lodging") ) { return "Hotel"; } if ( all.includes("veterinary_care") || all.includes("veterinarian") ) { return "Veterinary"; } if ( all.includes("pet_store") ) { return "Pet Shop"; } return "Other"; } function getPhotoName(place) { if (!Array.isArray(place.photos) || !place.photos.length) { return ""; } return String(place.photos[0]?.name || ""); } function getPetPolicy(place) { if (place.allowsDogs === true) { return { status: "confirmed", allowsDogs: true, text: "Dogs are indicated as allowed by Google Places." }; } if (place.allowsDogs === false) { return { status: "confirmed", allowsDogs: false, text: "Dogs are indicated as not allowed by Google Places." }; } return { status: "unknown", allowsDogs: null, text: "Pet policy is not confirmed. Contact the venue before visiting." }; } function buildOpeningHours(place) { const hours = place.currentOpeningHours || place.regularOpeningHours || null; if (!hours) { return null; } return { openNow: typeof hours.openNow === "boolean" ? hours.openNow : null, weekdayDescriptions: Array.isArray(hours.weekdayDescriptions) ? hours.weekdayDescriptions : [] }; } function buildGooglePlace(place) { const lat = toNumber(place.location?.latitude); const lng = toNumber(place.location?.longitude); if (lat === null || lng === null) { return null; } const petPolicy = getPetPolicy(place); return { id: `google-${place.id}`, providerId: place.id, source: "google", name: place.displayName?.text || "Pet Friendly Place", category: normalizeGoogleCategory(place), categoryLabel: place.primaryTypeDisplayName?.text || place.primaryType || "", address: place.formattedAddress || "", latitude: lat, longitude: lng, rating: typeof place.rating === "number" ? place.rating : null, ratingCount: Number.isFinite(Number(place.userRatingCount)) ? Number(place.userRatingCount) : 0, website: place.websiteUri || "", googleMapsUrl: place.googleMapsUri || "", phone: place.internationalPhoneNumber || place.nationalPhoneNumber || "", businessStatus: place.businessStatus || "", openingHours: buildOpeningHours(place), allowsDogs: petPolicy.allowsDogs, petPolicyStatus: petPolicy.status, petPolicy: petPolicy.text, photoName: getPhotoName(place), verified: false, trustSource: "Google Places" }; } async function fetchGooglePlaces({ latitude, longitude, radius, category }) { const apiKey = process.env.GOOGLE_PLACES_API_KEY; if (!apiKey) { return { enabled: false, places: [], error: "GOOGLE_PLACES_API_KEY is not configured." }; } const includedTypes = CATEGORY_TYPES[category] || CATEGORY_TYPES.All; const requestBody = { includedTypes, maxResultCount: MAX_RESULTS, rankPreference: "POPULARITY", locationRestriction: { circle: { center: { latitude, longitude }, radius } } }; const response = await fetch( GOOGLE_PLACES_URL, { method: "POST", headers: { "Content-Type": "application/json", "X-Goog-Api-Key": apiKey, "X-Goog-FieldMask": buildFieldMask() }, body: JSON.stringify(requestBody) } ); if (!response.ok) { const details = await response .text() .catch(() => ""); throw new Error( `Google Places request failed with ${response.status}${ details ? `: ${details}` : "" }` ); } const data = await response.json(); const places = Array.isArray(data.places) ? data.places .map(buildGooglePlace) .filter(Boolean) : []; return { enabled: true, places }; } function mergePlaces(...groups) { const unique = new Map(); groups .flat() .filter(Boolean) .forEach(place => { const key = place.providerId || place.id || [ String(place.name || "").toLowerCase(), Number(place.latitude).toFixed(5), Number(place.longitude).toFixed(5) ].join("|"); if (!unique.has(key)) { unique.set(key, place); } }); return [...unique.values()]; } export default async function handler(req, res) { setCors(res); if (req.method === "OPTIONS") { return res.status(204).end(); } if (req.method !== "POST") { return json( res, 405, { ok: false, error: "Method not allowed." } ); } try { const latitude = toNumber(req.body?.latitude); const longitude = toNumber(req.body?.longitude); if ( latitude === null || longitude === null || latitude < -90 || latitude > 90 || longitude < -180 || longitude > 180 ) { return json( res, 400, { ok: false, error: "Valid latitude and longitude are required." } ); } const radius = clampRadius(req.body?.radius); const category = normalizeCategory(req.body?.category); const google = await fetchGooglePlaces({ latitude, longitude, radius, category }); /* Later we can add PETS & DOGUE Verified places here from Supabase without exposing the database secret. Example future structure: const doguePlaces = await fetchVerifiedDoguePlaces({ latitude, longitude, radius, category }); const places = mergePlaces( doguePlaces, google.places ); */ const places = mergePlaces( google.places ); return json( res, 200, { ok: true, location: { latitude, longitude, radius }, category, sources: { google: { enabled: google.enabled, count: google.places.length }, dogue: { enabled: false, count: 0 } }, count: places.length, places } ); } catch (error) { console.error( "pet-friendly-places API error:", error ); return json( res, 500, { ok: false, error: "Unable to load nearby places right now." } ); } }
